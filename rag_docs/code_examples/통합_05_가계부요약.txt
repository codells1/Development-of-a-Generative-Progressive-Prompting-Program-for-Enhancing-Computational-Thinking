# 주제: 월별 가계부 요약 (통합/융합)
# 난이도: 고1~고2
# 사용 개념: 리스트, 딕셔너리, 함수, 반복문, 조건문, 누적 집계
# 통합 포인트: 유형별 집계 함수와 잔액 판정 출력 함수가 모여 한 달 가계부 요약을 만든다

def total_by_type(records, kind):
    total = 0
    for record in records:
        if record["type"] == kind:
            total += record["amount"]
    return total

def print_summary(records):
    income = total_by_type(records, "수입")
    expense = total_by_type(records, "지출")
    balance = income - expense
    print(f"수입 {income:,} / 지출 {expense:,} / 잔액 {balance:,}")
    if balance < 0:
        print("[주의] 적자입니다.")
    else:
        print("[양호] 흑자입니다.")

records = [{"type": "수입", "amount": 50000},
           {"type": "지출", "amount": 12000},
           {"type": "지출", "amount": 8000}]
print_summary(records)
