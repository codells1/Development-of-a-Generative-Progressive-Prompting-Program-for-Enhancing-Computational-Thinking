# 주제: 객관식 퀴즈 자동 채점 (통합/융합)
# 난이도: 중3~고1
# 사용 개념: 리스트, 딕셔너리, 함수, 반복문, 조건문
# 통합 포인트: 채점 함수와 점수·등급 출력 함수가 모여 한 번의 채점을 완성한다

def grade_quiz(quiz, answers):
    correct = 0
    for i in range(len(quiz)):
        if answers[i] == quiz[i]["answer"]:
            correct += 1
    return correct

def print_result(quiz, answers):
    correct = grade_quiz(quiz, answers)
    percent = correct / len(quiz) * 100
    level = "우수" if percent >= 70 else "노력요함"
    print(f"맞은 개수: {correct}/{len(quiz)}")
    print(f"점수: {percent:.1f}점 ({level})")

quiz = [{"answer": "B"}, {"answer": "A"}, {"answer": "C"}, {"answer": "D"}]
my_answers = ["B", "A", "B", "D"]
print_result(quiz, my_answers)
