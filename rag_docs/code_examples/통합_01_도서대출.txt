# 주제: 도서 대출 관리 시스템 (통합/융합)
# 난이도: 중3~고1
# 사용 개념: 딕셔너리, 리스트, 함수, 조건문, 반복문
# 통합 포인트: 도서 검색·대출·반납·현황 출력의 네 기능이 모여 하나의 대출 시스템을 이룬다

def find_book(library, title):
    for book in library:
        if book["title"] == title:
            return book
    return None

def checkout(library, title):
    book = find_book(library, title)
    if book is None:
        return f"'{title}'은(는) 목록에 없습니다."
    if book["available"] <= 0:
        return f"'{title}'은(는) 모두 대출 중입니다."
    book["available"] -= 1
    return f"'{title}' 대출 완료. 남은 수량 {book['available']}권."

def give_back(library, title):
    book = find_book(library, title)
    if book is None:
        return f"'{title}'은(는) 목록에 없습니다."
    book["available"] += 1
    return f"'{title}' 반납 완료. 현재 수량 {book['available']}권."

def print_status(library):
    print("== 도서 현황 ==")
    for book in library:
        state = "대출가능" if book["available"] > 0 else "대출불가"
        print(f"{book['title']}: {book['available']}/{book['total']}권 ({state})")
    print("-" * 25)

library = [
    {"title": "파이썬 입문", "total": 3, "available": 3},
    {"title": "알고리즘 기초", "total": 2, "available": 2},
    {"title": "자료구조", "total": 1, "available": 1},
]

requests = [
    ("대출", "파이썬 입문"),
    ("대출", "자료구조"),
    ("대출", "자료구조"),
    ("반납", "파이썬 입문"),
    ("대출", "없는책"),
]

for action, title in requests:
    if action == "대출":
        print(checkout(library, title))
    elif action == "반납":
        print(give_back(library, title))

print()
print_status(library)
