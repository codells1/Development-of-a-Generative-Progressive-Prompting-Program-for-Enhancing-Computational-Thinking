# 주제: 도서 대출 현황 (통합/융합)
# 난이도: 중3~고1
# 사용 개념: 딕셔너리, 리스트, 함수, 조건문, 반복문
# 통합 포인트: 재고 집계 함수와 현황 출력 함수가 모여 대출 가능 현황을 보여준다

def count_available(library):
    available = 0
    for book in library:
        if book["stock"] > 0:
            available += 1
    return available

def print_status(library):
    print(f"대출 가능: {count_available(library)}종 / 전체 {len(library)}종")
    for book in library:
        state = "가능" if book["stock"] > 0 else "불가"
        print(f"{book['title']}: {book['stock']}권 ({state})")

library = [{"title": "파이썬 입문", "stock": 3},
           {"title": "알고리즘 기초", "stock": 0},
           {"title": "자료구조", "stock": 1}]
print_status(library)
