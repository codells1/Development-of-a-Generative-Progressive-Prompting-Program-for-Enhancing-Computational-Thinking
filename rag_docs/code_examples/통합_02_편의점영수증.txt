# 주제: 편의점 영수증·거스름돈 계산 (통합/융합)
# 난이도: 중3~고1
# 사용 개념: 리스트, 딕셔너리, 함수, 반복문, 조건문, 산술 연산
# 통합 포인트: 합계·할인 계산 함수와 거스름돈 출력 함수가 모여 결제 한 건을 처리한다

def total_with_discount(cart):
    total = 0
    for item in cart:
        total += item["price"] * item["qty"]
    if total >= 10000:
        total = total * 0.95
    return total

def print_receipt(cart, paid):
    pay = total_with_discount(cart)
    print(f"받을 돈: {pay:,.0f}원")
    print(f"거스름돈: {paid - pay:,.0f}원")

cart = [{"name": "도시락", "price": 4500, "qty": 2},
        {"name": "음료", "price": 1500, "qty": 1}]
paid = 12000
print_receipt(cart, paid)
